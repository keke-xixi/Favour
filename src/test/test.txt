<template>
  <view class="mine-car-config">
    <!-- 日期选择器 -->
    <u-calendar 
      ref="calendar"
      v-model="showDate"
      mode="date" 
      @change="changeDate" 
      @close="closeDate" 
      :defaultDate="defaultDateMultiple" 
    ></u-calendar>
    
    <!-- 页面标题 -->
    <view class="page-title">矿车关联配置</view>

    <!-- 矿场卡片列表 -->
    <view class="card-list">
      <view class="config-card" v-for="mine in mineList" :key="mine.mineId">
        <!-- 日期和车牌配置区域 -->
        <view class="card-body">
          <!-- 日期配置区域 -->
          <view class="date-section">
            <view class="section-header">
              <!-- 矿的名称放在这里 -->
              <text class="mine-title">{{ mine.mineShortName }}</text>
              <view class="add-date-btn-right" @click="addDate(mine.mineId)">
                <view class="add-btn-content">
                  <view class="add-icon">+</view>
                  <text class="add-text">添加日期</text>
                </view>
              </view>
            </view>
            
            <!-- 日期列表 -->
            <view class="date-list">
              <view 
                class="date-item" 
                v-for="(dateConfig, dateIndex) in mine.dateConfigs" 
                :key="dateIndex"
              >
                <!-- 日期标题 -->
                <view class="date-header">
                  <view class="date-tag">
                    <text class="date-text">{{ dateConfig.date }}</text>
                  </view>
                  <button 
                    @click.stop="handleDeleteDate(mine, dateIndex)" 
                    class="delete-btn"
                  >×</button>
                </view>
                
                <!-- 该日期下的车牌 -->
                <view class="car-list">
                  <view 
                    class="tag" 
                    v-for="(car, carIndex) in dateConfig.cars" 
                    :key="carIndex"
                  >
                    <text class="tag-text">{{ car.carNo }}</text>
                    <button 
                      @click.stop="handleDeleteCar(mine, dateIndex, carIndex)" 
                      class="delete-btn"
                    >×</button>
                  </view>
                  
                  <view 
                    class="tag add-tag" 
                    @click="handleOpenCarPopup(mine.mineId, dateIndex)"
                  >
                    <text class="tag-text">添加车牌</text>
                    <view class="add-increase-btn">+</view>
                  </view>
                </view>
              </view>
            </view>
          </view>
        </view>

        <!-- 保存按钮 -->
        <view class="card-footer">
          <u-button 
            type="primary"
            @click="handleSave(mine)"
            style="height: 80rpx; width: 100%;"
          >
            保存配置
          </u-button>
        </view>
      </view>
    </view>

    <!-- 车牌选择弹出层 -->
    <view class="popup-mask" v-show="showCarPopup" @click="showCarPopup = false"></view>
    <view 
      class="popup-container car-popup" 
      v-show="showCarPopup"
      :class="{ 'popup-show': showCarPopup }"
    >
      <view class="popup-header">
        <text class="popup-title">选择车牌</text>
        <view>
          <button type="primary" @click="completeSelect" class="add-btn">完成</button>
        </view>
      </view>

      <!-- 搜索框 -->
      <view class="popup-search">
        <view class="icon-search"></view>
        <input
          v-model="carSearchVal"
          placeholder="搜索车牌（如：湘A、123）"
          class="search-input"
          @input="filterCarList"
        />
        <button 
          @click="carSearchVal = ''" 
          class="clear-btn"
          v-if="carSearchVal"
        >
          <view class="icon-close"></view>
        </button>
      </view>

      <!-- 车牌列表 -->
      <scroll-view class="popup-list car-list" scroll-y>
        <view class="no-result" v-if="filteredCarList.length === 0">
          暂无匹配车牌
        </view>
        <view 
          class="popup-item car-item" 
          v-for="car in filteredCarList" 
          :key="car.id"
          @click="handleSelectCar(car.carNo)"
        >
          <text class="item-text">{{ car.carNo }}</text>
          <view class="icon-check" v-if="tempSelectedCars.includes(car.carNo)"></view>
        </view>
      </scroll-view>
    </view>
  </view>
</template>

<script>
export default {
  name: 'MineCarConfig',
  data() {
    return {
      showDate: false,
      mode: 'multiple',
      defaultDateMultiple: [],
      
      // 矿场列表数据
      mineList: [
        { 
          mineId: '1',
          mineNo: 'K007',
          mineShortName: 'A矿',
          mineFullName: 'A矿',
          dateConfigs: [
            {
              date: '2025-01-02',
              cars: [
                { carID: '1', carNo: '湘A12345' },
                { carID: '2', carNo: '湘A99999' }
              ]
            },
            {
              date: '2025-01-03',
              cars: [
                { carID: '3', carNo: '粤B56789' }
              ]
            }
          ]
        },
        { 
          mineId: '2',
          mineNo: 'K008',
          mineShortName: 'B矿',
          mineFullName: 'B矿',
          dateConfigs: [
            {
              date: '2025-01-02',
              cars: [
                { carID: '4', carNo: '鄂A11223' }
              ]
            }
          ]
        }
      ],
      
      // 车牌库
      allCarList: [
        { carID: '1', carNo: '湘A12345' },
        { carID: '2', carNo: '湘A99999' },
        { carID: '3', carNo: '粤B56789' },
        { carID: '4', carNo: '鄂A11223' },
        { carID: '5', carNo: '湘A33445' },
        { carID: '6', carNo: '粤B67890' },
        { carID: '7', carNo: '鄂A55667' },
        { carID: '8', carNo: '湘A77889' },
        { carID: '9', carNo: '粤B90123' },
        { carID: '10', carNo: '鄂A23456' }
      ],
      
      showCarPopup: false,
      carSearchVal: '',
      filteredCarList: [],
      currentMineId: '',
      currentDateIndex: -1,
      tempSelectedCars: [] // 临时存储当前选中的车牌
    };
  },
  created() {
    // 初始化过滤车牌列表
    this.filteredCarList = [...this.allCarList];
  },
  watch: {
    carSearchVal(val) {
      if (!val) {
        this.filteredCarList = [...this.allCarList];
      }
    }
  },
  methods: {
    // 确认日期选择
    changeDate(e) {
      const { result } = e;
      const mine = this.mineList.find(i => i.mineId === this.currentMineId);
      
      // 检查该日期是否已存在
      const dateExists = mine.dateConfigs.some(config => config.date === result);
      
      if (!dateExists) {
        // 添加新日期配置
        mine.dateConfigs.push({
          date: result,
          cars: []
        });
      }
      
      this.showDate = false;
    },
    
    // 关闭日期选择器
    closeDate() {
      this.showDate = false;
    },
    
    // 添加日期
    addDate(mineId) {
      this.currentMineId = mineId;
      this.showDate = true;
    },
    
    // 打开车牌选择弹出层
    handleOpenCarPopup(mineId, dateIndex) {
      this.currentMineId = mineId;
      this.currentDateIndex = dateIndex;
      this.carSearchVal = '';
      this.filteredCarList = [...this.allCarList];
      
      // 初始化临时选中的车牌
      const mine = this.mineList.find(i => i.mineId === mineId);
      this.tempSelectedCars = mine.dateConfigs[dateIndex].cars.map(car => car.carNo);
      
      this.showCarPopup = true;
    },
    
    // 模糊搜索过滤车牌
    filterCarList() {
      const keyword = this.carSearchVal.trim().toLowerCase();
      this.filteredCarList = this.allCarList.filter(car => 
        car.carNo.toLowerCase().includes(keyword)
      );
    },
    
    // 选择/取消车牌
    handleSelectCar(carNo) {
      const index = this.tempSelectedCars.indexOf(carNo);
      if (index > -1) {
        this.tempSelectedCars.splice(index, 1); // 取消选择
      } else {
        this.tempSelectedCars.push(carNo); // 新增选择
      }
    },
    
    // 完成车牌选择
    completeSelect() {
      const mine = this.mineList.find(i => i.mineId === this.currentMineId);
      
      // 更新该日期下的车牌
      mine.dateConfigs[this.currentDateIndex].cars = this.tempSelectedCars.map(carNo => {
        // 从车牌库中找到对应的车牌对象
        return this.allCarList.find(car => car.carNo === carNo);
      }).filter(car => car); // 过滤掉未找到的车牌
      
      this.showCarPopup = false;
    },
    
    // 删除车牌
    handleDeleteCar(mine, dateIndex, carIndex) {
      mine.dateConfigs[dateIndex].cars.splice(carIndex, 1);
    },
    
    // 删除日期配置
    handleDeleteDate(mine, dateIndex) {
      mine.dateConfigs.splice(dateIndex, 1);
    },
    
    // 保存配置
    handleSave(mine) {
      console.log('保存矿场配置:', mine);
      uni.showToast({
        title: "保存成功",
        icon: 'success'
      });
    }
  }
};
</script>
