flex布局 :
justify-items: start; 换行后从新行的起始位置开始（默认最左侧）
align-items: center;  竖直方向居中
gap: 10px;  /* 设置盒子之间的间隔 */
.container {
    display: flex;
    flex-wrap: wrap; /* 允许换行 */
    gap: 20px; /* 设置盒子之间的间隔 */
}

.box {
    flex: 1 1 calc(33.33% - 20px); /* 每个盒子占据三分之一的宽度，减去间隔 */
    background-color: lightblue; /* 仅供视觉效果 */
    padding: 20px; /* 添加内边距 */
    text-align: center; /* 内容居中 */
}
