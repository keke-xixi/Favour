// 根据窗口大小调整位置计算
const calcPosition = (percent) => {
  const containerWidth = container.value.offsetWidth
  const containerHeight = container.value.offsetHeight
  return {
    x: (percent[0] / 100) * containerWidth,
    y: (percent[1] / 100) * containerHeight
  }
}
